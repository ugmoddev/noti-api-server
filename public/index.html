Hướng dẫn sử dụng

1. Push Notification

curl -X POST https://your-api.onrender.com/api/push \
  -H "Content-Type: application/json" \
  -d '{
    "id": "b2b24895f69e",
    "apiKey": "87ebc4e597aab91a0aae0900ee6c6753bdb6527cf02d8aa18bf617bf57a57005",
    "job": "game_job_id",
    "players": 5,
    "sea": 2,
    "boss": "darkbeard"
  }'

2. Get Notifications

# Lấy tất cả
curl https://your-api.onrender.com/api/notifications

# Lọc theo Sea
curl https://your-api.onrender.com/api/notifications?sea=3

# Lọc theo Boss
curl https://your-api.onrender.com/api/notifications?boss=darkbeard
